mission "Gegno Asteroid Mining - Aluminum"
	job
	repeat
	name "采集10吨铝"
	description "开采并收集10吨铝，然后将矿物带回至 <origin> 的格诺处。"
	source "Tschyss"
	to offer
		has "Gegno Asteroid Mining Prologue: done"
		random < 10
	to accept
		"cargo space" > 10
	on visit
		dialog "你已回到了 <origin>，但你的货舱中并没有10吨铝。降落前，请确保已开采并收集10吨铝装入货舱。"
	on complete
		"gegno: mining jobs" ++
		payment 36000
		outfit "Aluminum" -10
		dialog phrase "gegno mining payment dialog"
